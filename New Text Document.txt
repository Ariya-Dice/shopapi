server {
    listen 80;
    server_name arbshop.ir www.arbshop.ir;

    return 301 https://arbshop.ir$request_uri;
}

server {
    listen 443 ssl;
    server_name arbshop.ir www.arbshop.ir;

    ssl_certificate /etc/letsencrypt/live/arbshop.ir/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/arbshop.ir/privkey.pem;

    location /api/ {
        proxy_pass http://127.0.0.1:3000;

        proxy_http_version 1.1;

        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
    }

    location / {
        root /var/www/rbshop;
        try_files $uri $uri/ /index.html;
    }
}
